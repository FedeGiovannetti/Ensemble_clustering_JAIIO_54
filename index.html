3. Selected algorithms comparison

A partir de lo anterior, termino quedándome con tres métodos (kmeans, PAM ecludieano y PAM manhattan). Entonces, ahora me va a interesar ver cómo son los resultados de cada método en términos del tamaño de cada grupo y la cantidad de diferencias entre si para cada modelo. Es decir, me interesa tener grupos lo más homogeneos posibles y que sean lo más diferentes posible entre si.

Esto lo hacemos con dos medidas de consenso distintas: Majority voting (que es la clásica pero menos robusta), y LCE que es más robusta.

algos_info <- function(data, algorithms, cons_funct){
  
  model_names <- c()

  group_n <- c()
  
  significant_p <- c()
  
  for (i in 1:length(algorithms)) {
    
    model_names[i] = names(algorithms)[i]
    
    datos_prueba = data
    
    if (cons_funct == "majority") {
      
      datos_prueba$Clusters = majority_voting(algorithms[[i]])
      
    }else{
      
      datos_prueba$Clusters = LCE(algorithms[[i]], 3)
      
    }
  
    
  
    group_n[i] <- paste(as.vector(table(datos_prueba$Clusters)), collapse = " ")
    
  
    # Transform to long format
    
    datos.long.z = datos_prueba %>%
      pivot_longer(!Clusters,names_to = "Feature") %>% 
      mutate(Tipo = ifelse(grepl("rt", Feature), "RT", "Desempeño")) %>% 
      feature_rename() %>% 
      arrange(factor(Feature, levels = feature_order)) 
    
    
    # Inferential analysis
    
    datos.kruskal.z = datos.long.z %>%
      group_by(Feature) %>% 
      nest() %>% 
      mutate(modelos = map(data ,~kruskal.test(value ~ Clusters, data = .)),
             summary = map(modelos, broom::tidy),) %>% 
      unnest(summary) %>% 
      select(-data,-modelos, -parameter, -method) 
    
    significant_p[i] <- datos.kruskal.z %>% 
      filter(p.value < .05) %>% 
      nrow()

  
}


data.frame(model_names, group_n, significant_p)
  
  
}
